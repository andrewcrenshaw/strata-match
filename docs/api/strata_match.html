strata_match

strata-match: Two-stage vector + LLM job-to-profile matching.

This package exposes a small public API for matching job postings to candidate profiles:

  • Factory: create_matcher() returns a Matcher configured with embedding and optional LLM scoring backends.
  • Matching: match_job() and match_batch() score jobs against a profile.
  • Data types: CandidateProfile, JobDescription, MatchResult, BatchMatchResult, and ConfidenceTier.

Advanced integrations (custom embedding/LLM providers) live under strata_match.embeddings, strata_match.llm, and strata_match.providers — they are not re-exported from the root package.

Example::

from strata_match import (
    CandidateProfile,
    JobDescription,
    create_matcher,
    match_job,
)

matcher = create_matcher("openai", vector_threshold=0.35)
profile = CandidateProfile(title="Engineer", skills=["Python"])
job = JobDescription(title="Backend Role", company="Acme")
result = await match_job(matcher, profile, job)
print(result.score, result.confidence_tier)

 37class Matcher:
 38    """Two-stage matching engine.
 39
 40    **Stage 1 (vector):** Cosine similarity between profile and job embeddings
 41    (or pre-computed vectors on the models). Produces a score in ``[0, 100]``.
 42
 43    **Stage 2 (LLM):** When ``llm_scorer`` is set and the Stage 1 raw score is
 44    at or above ``vector_threshold``, the job is sent to the LLM for nuanced
 45    scoring (rationale, strengths, gaps). Below the threshold, Stage 2 is
 46    skipped and confidence is typically LOW.
 47
 48    Args:
 49        vector_scorer: Stage-1 scorer wrapping an embedding provider.
 50        vector_threshold: Minimum raw cosine similarity in ``[0, 1]`` before
 51            Stage 2 runs (when an LLM scorer is configured).
 52        llm_scorer: Optional Stage-2 scorer; ``None`` means vector-only matching.
 53        max_concurrency: Maximum concurrent LLM scoring calls in :meth:`match_batch`
 54            (when ``llm_scorer`` is set). Values below ``1`` are treated as ``1``.
 55
 56    Example::
 57
 58        matcher = create_matcher("openai", vector_threshold=0.35)
 59        result = await matcher.match_one(profile, job)
 60    """
 61
 62    vector_scorer: VectorScorer
 63    vector_threshold: float = 0.3
 64    llm_scorer: LLMScorer | None = None
 65    llm_confirm_threshold: float = 70.0
 66    max_concurrency: int = 5
 67
 68    async def match_one(self, profile: CandidateProfile, job: JobDescription) -> MatchResult:
 69        """Score a single job against the candidate profile.
 70
 71        Args:
 72            profile: Candidate data (skills, summary, optional embedding).
 73            job: Job posting data (title, requirements, optional embedding).
 74
 75        Returns:
 76            A :class:`~strata_match.models.MatchResult` with score, tier, and
 77            optional LLM fields.
 78
 79        Raises:
 80            StrataMatchError: Propagates embedding or LLM provider errors (network,
 81                authentication, rate limits) from the configured backends.
 82
 83        Example::
 84
 85            r = await matcher.match_one(profile, job)
 86            print(r.score, r.confidence_tier, r.rationale)
 87        """
 88        raw_score = await self.vector_scorer.score(profile, job)
 89        score_100 = _to_score_100(raw_score)
 90
 91        if raw_score < self.vector_threshold:
 92            return build_match_result(
 93                job,
 94                score=score_100,
 95                vector_score=score_100,
 96                confidence_tier=ConfidenceTier.LOW,
 97            )
 98
 99        if self.llm_scorer is not None:
100            llm_result = await self.llm_scorer.score(profile, job, vector_score=score_100)
101            confidence = classify_confidence(
102                raw_score,
103                llm_confirmed=llm_result.score >= self.llm_confirm_threshold,
104            )
105            return MatchResult(
106                job_title=llm_result.job_title,
107                job_company=llm_result.job_company,
108                score=llm_result.score,
109                vector_score=score_100,
110                confidence_tier=confidence,
111                rationale=llm_result.rationale,
112                strengths=llm_result.strengths,
113                gaps=llm_result.gaps,
114                salary_match=llm_result.salary_match,
115                culture_signals=llm_result.culture_signals,
116                llm_scored=llm_result.llm_scored,
117                llm_error=llm_result.llm_error,
118                tokens_used=llm_result.tokens_used,
119            )
120
121        confidence = classify_confidence(raw_score, llm_confirmed=False)
122        return build_match_result(
123            job,
124            score=score_100,
125            vector_score=score_100,
126            confidence_tier=confidence,
127            llm_scored=False,
128        )

130    async def match_batch(
131        self, profile: CandidateProfile, jobs: list[JobDescription]
132    ) -> BatchMatchResult:
133        """Score multiple jobs against the candidate profile.
134
135        Jobs below ``vector_threshold`` are counted in ``jobs_skipped`` and do
136        not receive LLM scoring. Remaining results are sorted by ``score``
137        descending.
138
139        Args:
140            profile: Candidate data shared across all jobs.
141            jobs: List of job postings to evaluate (may be empty).
142
143        Returns:
144            :class:`~strata_match.models.BatchMatchResult` with per-job
145            results, counts, and token totals.
146
147        Raises:
148            StrataMatchError: Propagates embedding or LLM provider errors from the
149                configured backends.
150
151        Example::
152
153            batch = await matcher.match_batch(profile, open_jobs)
154            for r in batch.results:
155                print(r.job_title, r.score)
156        """
157        if not jobs:
158            return BatchMatchResult(
159                results=[],
160                jobs_evaluated=0,
161                jobs_skipped=0,
162                total_tokens=0,
163                duration_ms=0.0,
164                llm_scored_count=0,
165            )
166
167        start = time.perf_counter()
168        scores = await self.vector_scorer.score_batch(profile, jobs)
169
170        results: list[MatchResult] = []
171        skipped = 0
172        llm_count = 0
173        llm_fallback_count = 0
174        total_tokens = 0
175
176        llm_pairs: list[tuple[JobDescription, float]] = []
177
178        for job, raw in zip(jobs, scores, strict=True):
179            if raw < self.vector_threshold:
180                skipped += 1
181                continue
182
183            score_100 = _to_score_100(raw)
184
185            if self.llm_scorer is not None:
186                llm_pairs.append((job, raw))
187            else:
188                confidence = classify_confidence(raw, llm_confirmed=False)
189                results.append(
190                    build_match_result(
191                        job,
192                        score=score_100,
193                        vector_score=score_100,
194                        confidence_tier=confidence,
195                        llm_scored=False,
196                    )
197                )
198
199        if self.llm_scorer is not None and llm_pairs:
200            cap = max(1, self.max_concurrency)
201            semaphore = asyncio.Semaphore(cap)
202            llm = self.llm_scorer
203
204            async def _score_one_llm(job: JobDescription, raw: float) -> MatchResult:
205                score_100 = _to_score_100(raw)
206                async with semaphore:
207                    return await llm.score(profile, job, vector_score=score_100)
208
209            gathered = await asyncio.gather(
210                *(_score_one_llm(j, r) for j, r in llm_pairs),
211                return_exceptions=True,
212            )
213
214            for (job, raw), item in zip(llm_pairs, gathered, strict=True):
215                score_100 = _to_score_100(raw)
216                if isinstance(item, BaseException):
217                    # AC2: preserve the row as a fallback instead of dropping
218                    logger.error(
219                        "match_batch: LLM scoring error for job %r; returning fallback row",
220                        job.title,
221                        exc_info=(type(item), item, item.__traceback__),
222                    )
223                    results.append(
224                        build_match_result(
225                            job,
226                            score=score_100,
227                            vector_score=score_100,
228                            llm_scored=False,
229                            llm_error=f"LLM scoring error: {item}",
230                        )
231                    )
232                    llm_fallback_count += 1  # AC3: distinct counter
233                    continue
234
235                llm_result = item
236                score_100 = _to_score_100(raw)
237                confidence = classify_confidence(
238                    raw,
239                    llm_confirmed=llm_result.score >= self.llm_confirm_threshold,
240                )
241                results.append(
242                    MatchResult(
243                        job_title=llm_result.job_title,
244                        job_company=llm_result.job_company,
245                        score=llm_result.score,
246                        vector_score=score_100,
247                        confidence_tier=confidence,
248                        rationale=llm_result.rationale,
249                        strengths=llm_result.strengths,
250                        gaps=llm_result.gaps,
251                        salary_match=llm_result.salary_match,
252                        culture_signals=llm_result.culture_signals,
253                        llm_scored=llm_result.llm_scored,
254                        llm_error=llm_result.llm_error,
255                        tokens_used=llm_result.tokens_used,
256                    )
257                )
258                if llm_result.llm_scored:
259                    llm_count += 1
260                elif llm_result.llm_error is not None:
261                    llm_fallback_count += 1  # AC3: LLMScorer returned a fallback
262                total_tokens += llm_result.tokens_used
263
264        results.sort(key=lambda r: r.score, reverse=True)
265
266        elapsed_ms = (time.perf_counter() - start) * 1000
267        return BatchMatchResult(
268            results=results,
269            jobs_evaluated=len(jobs),
270            jobs_skipped=skipped,
271            total_tokens=total_tokens,
272            duration_ms=round(elapsed_ms, 2),
273            llm_scored_count=llm_count,
274            llm_fallback_count=llm_fallback_count,
275        )

131        self, profile: CandidateProfile, jobs: list[JobDescription]
132    ) -> BatchMatchResult:
133        """Score multiple jobs against the candidate profile.
134
135        Jobs below ``vector_threshold`` are counted in ``jobs_skipped`` and do
136        not receive LLM scoring. Remaining results are sorted by ``score``
137        descending.
138

140            profile: Candidate data shared across all jobs.
141            jobs: List of job postings to evaluate (may be empty).
142

144            :class:`~strata_match.models.BatchMatchResult` with per-job
145            results, counts, and token totals.
146
147        Raises:
148            StrataMatchError: Propagates embedding or LLM provider errors from the
149                configured backends.
150
151        Example::
152
153            batch = await matcher.match_batch(profile, open_jobs)
154            for r in batch.results:
155                print(r.job_title, r.score)
156        """
157        if not jobs:
158            return BatchMatchResult(
159                results=[],
160                jobs_evaluated=0,
161                jobs_skipped=0,
162                total_tokens=0,
163                duration_ms=0.0,
164                llm_scored_count=0,
165            )
166
167        start = time.perf_counter()
168        scores = await self.vector_scorer.score_batch(profile, jobs)
169
170        results: list[MatchResult] = []
171        skipped = 0
172        llm_count = 0
173        llm_fallback_count = 0
174        total_tokens = 0
175
176        llm_pairs: list[tuple[JobDescription, float]] = []
177
178        for job, raw in zip(jobs, scores, strict=True):
179            if raw < self.vector_threshold:
180                skipped += 1
181                continue
182
183            score_100 = _to_score_100(raw)
184
185            if self.llm_scorer is not None:
186                llm_pairs.append((job, raw))
187            else:
188                confidence = classify_confidence(raw, llm_confirmed=False)
189                results.append(
190                    build_match_result(
191                        job,
192                        score=score_100,
193                        vector_score=score_100,
194                        confidence_tier=confidence,
195                        llm_scored=False,
196                    )
197                )
198
199        if self.llm_scorer is not None and llm_pairs:
200            cap = max(1, self.max_concurrency)
201            semaphore = asyncio.Semaphore(cap)
202            llm = self.llm_scorer
203
204            async def _score_one_llm(job: JobDescription, raw: float) -> MatchResult:
205                score_100 = _to_score_100(raw)
206                async with semaphore:
207                    return await llm.score(profile, job, vector_score=score_100)
208
209            gathered = await asyncio.gather(
210                *(_score_one_llm(j, r) for j, r in llm_pairs),
211                return_exceptions=True,
212            )
213

215                score_100 = _to_score_100(raw)
216                if isinstance(item, BaseException):
217                    # AC2: preserve the row as a fallback instead of dropping
218                    logger.error(

223                    results.append(
224                        build_match_result(
225                            job,
226                            score=score_100,
227                            vector_score=score_100,
228                            llm_scored=False,
229                            llm_error=f"LLM scoring error: {item}",
230                        )
231                    )
232                    llm_fallback_count += 1  # AC3: distinct counter
233                    continue
234
235                llm_result = item
236                score_100 = _to_score_100(raw)
237                confidence = classify_confidence(
238                    raw,
239                    llm_confirmed=llm_result.score >= self.llm_confirm_threshold,
240                )
241                results.append(
242                    MatchResult(
243                        job_title=llm_result.job_title,
244                        job_company=llm_result.job_company,
245                        score=llm_result.score,
246                        vector_score=score_100,
247                        confidence_tier=confidence,
248                        rationale=llm_result.rationale,
249                        strengths=llm_result.strengths,
250                        gaps=llm_result.gaps,
251                        salary_match=llm_result.salary_match,
252                        culture_signals=llm_result.culture_signals,
253                        llm_scored=llm_result.llm_scored,
254                        llm_error=llm_result.llm_error,
255                        tokens_used=llm_result.tokens_used,
256                    )
257                )
258                if llm_result.llm_scored:
259                    llm_count += 1
260                elif llm_result.llm_error is not None:
261                    llm_fallback_count += 1  # AC3: LLMScorer returned a fallback
262                total_tokens += llm_result.tokens_used
263
264        results.sort(key=lambda r: r.score, reverse=True)
265
266        elapsed_ms = (time.perf_counter() - start) * 1000
267        return BatchMatchResult(
268            results=results,
269            jobs_evaluated=len(jobs),
270            jobs_skipped=skipped,
271            total_tokens=total_tokens,
272            duration_ms=round(elapsed_ms, 2),
273            llm_scored_count=llm_count,
274            llm_fallback_count=llm_fallback_count,
275        )
